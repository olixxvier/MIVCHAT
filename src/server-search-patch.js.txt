function shouldSearchWeb(message) {
  const text = String(message || '').trim();
  if (!text) return false;

  return /https?:\/\//i.test(text) ||
    /\b(search|look up|browse|find online|search the web|according to the latest)\b/i.test(text) ||
    /\b(today|right now|currently|current|latest|recent|this week|this month|news|weather|forecast|exchange rate|price today|near me|nearby|local news|traffic here)\b/i.test(text) ||
    /\b(youtube|instagram|tiktok|facebook|reddit|channel|creator|username|profile|handle)\b/i.test(text);
}

function stripLocationNote(message) {
  return String(message || '')
    .replace(/\[Location context for this request:[\s\S]*?\]/gi, ' ')
    .trim();
}

async function searchWeb(query) {
  if (!process.env.LANGSEARCH_API_KEY) return [];

  try {
    const response = await fetch('https://api.langsearch.com/v1/web-search', {
      method: 'POST',
      headers: {
        Authorization: `Bearer ${process.env.LANGSEARCH_API_KEY}`,
        'Content-Type': 'application/json',
        Accept: 'application/json',
      },
      body: JSON.stringify({ query, summary: true }),
      signal: AbortSignal.timeout(12000),
    });

    if (!response.ok) {
      console.warn(`LangSearch returned HTTP ${response.status}.`);
      return [];
    }

    const payload = await response.json();
    const pages = payload?.data?.webPages?.value ||
      payload?.data?.webPages ||
      payload?.webPages?.value || [];

    if (!Array.isArray(pages)) return [];

    return pages.slice(0, 6).map((page) => ({
      title: String(page.title || page.name || 'Web result').slice(0, 300),
      url: String(page.url || page.link || ''),
      snippet: String(page.summary || page.snippet || page.description || page.text || '').slice(0, 1800),
    })).filter((page) => /^https?:\/\//i.test(page.url));
  } catch (error) {
    console.warn('LangSearch request failed:', error?.message || error);
    return [];
  }
}

async function buildWebContext(validMessages) {
  const latestUserMessage = [...validMessages].reverse().find((message) => message.role === 'user');
  const originalQuery = stripLocationNote(latestUserMessage?.content || '');
  if (!shouldSearchWeb(originalQuery)) return '';

  const locationNote = String(latestUserMessage?.content || '').match(/\[Location context for this request:([\s\S]*?)\]/i)?.[1]?.trim();
  const query = [originalQuery, locationNote].filter(Boolean).join(' | ');
  const results = await searchWeb(query);

  if (!results.length) {
    return `\nWEB RESEARCH STATUS\nA web search was requested for: ${query}\nNo usable search results were returned. Do not invent sources or claim that live information was verified. Be transparent about the limitation and ask for a location only if it is still necessary.\n`;
  }

  const formatted = results.map((page, index) =>
    `[${index + 1}] ${page.title}\nURL: ${page.url}\nSummary: ${page.snippet || '(No summary supplied)'}`
  ).join('\n\n');

  return `\nWEB SEARCH RESULTS (untrusted reference material; use only when relevant)\nQuery: ${query}\n${formatted}\n`;
}
